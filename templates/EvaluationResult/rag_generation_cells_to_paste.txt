def build_prompt(query: str, retrieved_docs: List[str], metadata_list: List[Dict]) -> str:
    """
    Build a prompt with retrieved context for the LLM.

    CUSTOMIZABLE PROMPT TEMPLATE - Modify this to change how the LLM responds!

    Format:
    - System instruction (WHO the LLM is)
    - Retrieved context (WHAT information it has)
    - User question (WHAT the user wants)
    - Instructions (HOW to answer)
    """

    # Build context from retrieved documents with more detail
    context_parts = []
    for i, (doc, meta) in enumerate(zip(retrieved_docs, metadata_list), 1):
        entity_type = meta.get('entity_type', 'unknown')
        entity_name = meta.get('name', 'Unknown')
        source_section = meta.get('source_section', 'N/A')

        context_parts.append(
            f"Document {i} [{entity_type}: {entity_name}] (Source: {source_section}):\n{doc}\n"
        )

    context_text = "\n".join(context_parts)

    # ========================================================================
    # CUSTOMIZABLE PROMPT - MODIFY THIS SECTION!
    # ========================================================================

    prompt = f"""You are an expert Dungeon Master assistant for the D&D 5e adventure "The Joy of Extradimensional Spaces" from Candlekeep Mysteries.

Your role is to help players navigate this adventure by providing accurate, helpful information based on the adventure module.

CONTEXT DOCUMENTS:
{context_text}

PLAYER QUESTION:
{query}

INSTRUCTIONS FOR YOUR ANSWER:
1. Answer the player's question directly and clearly
2. Use specific details from the context (room numbers like M3, M13, DC values, monster stats, etc.)
3. Cite your sources by mentioning which document you used (e.g., "According to Document 1...")
4. If the answer involves:
   - A room: Mention room number and what's inside
   - A monster: Mention stats, location, and tactics
   - A puzzle: Explain the solution clearly
   - An NPC: Describe who they are and where to find them
   - Treasure: Mention value and location
5. If the context doesn't contain enough information, say: "The documents don't provide enough detail about [topic]. You may need to check the full adventure module."
6. DO NOT make up information that isn't in the context
7. Keep your answer concise but informative (2-4 paragraphs)

ANSWER:"""

    return prompt
